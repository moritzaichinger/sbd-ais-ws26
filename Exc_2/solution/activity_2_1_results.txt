-- D. The top 5 customers by total spending.
-- The number of orders also helps us answer E.
SELECT customer_name,
       SUM(quantity * price_per_unit) AS total_spending,
       COUNT(*) AS number_of_orders
FROM orders
GROUP BY customer_name
ORDER BY total_spending DESC, customer_name ASC
LIMIT 5;

-- E. Check how often customer names are repeated.
SELECT COUNT(*) AS total_orders,
       COUNT(DISTINCT customer_name) AS different_customer_names,
       ROUND(COUNT(*)::numeric / COUNT(DISTINCT customer_name), 2)
         AS average_orders_per_name
FROM orders;
 total_orders | different_customer_names | average_orders_per_name 
--------------+--------------------------+-------------------------
      1000000 |                     1056 |                  946.97
(1 row)

-- E. Explanation:
-- The top customers have around 1,000 orders each and very high spending totals.
-- In ecommerce/dataset_generator.py, the names are randomly picked from
-- 32 first names and 33 last names. This gives only 32 * 33 = 1056 possible
-- full names for 1,000,000 orders.
-- So the same names appear many times, on average about 947 times each.
-- Grouping by name adds all these orders together, which explains the high
-- totals. The data is randomly generated and does not represent real customers.
